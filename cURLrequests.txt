# ------------------------------------------
# 1. ALBUM JDBC CONTROLLERS
# ------------------------------------------

# Create Album (Raw JDBC)
curl -X POST http://localhost:8080/api/albums/jdbc/raw \
  -H "Content-Type: application/json" \
  -d '{
    "title": "Raw Album Title",
    "releaseYear": 2024,
    "artistId": "00000000-0000-0000-0000-000000000000"
  }'

# Get All Albums (Raw JDBC)
curl -X GET http://localhost:8080/api/albums/jdbc/raw

# Get Album by ID (Raw JDBC)
curl -X GET http://localhost:8080/api/albums/jdbc/raw/00000000-0000-0000-0000-000000000000

# Update Album (Raw JDBC)
curl -X PUT http://localhost:8080/api/albums/jdbc/raw/00000000-0000-0000-0000-000000000000 \
  -H "Content-Type: application/json" \
  -d '{
    "title": "Updated Raw Album",
    "releaseYear": 2025,
    "artistId": "00000000-0000-0000-0000-000000000000"
  }'

# Delete Album (Raw JDBC)
